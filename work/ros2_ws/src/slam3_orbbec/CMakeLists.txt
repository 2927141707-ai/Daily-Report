cmake_minimum_required(VERSION 3.16)
project(slam3_orbbec LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(ament_cmake REQUIRED)
foreach(dep rclcpp sensor_msgs geometry_msgs nav_msgs std_msgs cv_bridge message_filters std_srvs tf2_ros)
 find_package(${dep} REQUIRED)
endforeach()
find_package(OpenCV 4 REQUIRED)
find_package(Eigen3 REQUIRED)
find_package(Pangolin REQUIRED)
set(ORB_SLAM3_ROOT "$ENV{ORB_SLAM3_ROOT}" CACHE PATH "ORB-SLAM3 source and build root")
if(NOT EXISTS "${ORB_SLAM3_ROOT}/lib/libORB_SLAM3.so")
 message(FATAL_ERROR "Source outputs/setup_slam3.sh first")
endif()
add_executable(rgbd_node src/rgbd_node.cpp)
# Match the local ORB-SLAM3/g2o Eigen vectorization ABI (-march=native).
# Mixing generic SSE and native AVX inline allocators can corrupt graph deletion.
target_compile_options(rgbd_node PRIVATE -march=native)
target_compile_definitions(rgbd_node PRIVATE EIGEN_MAX_ALIGN_BYTES=16 EIGEN_MAX_STATIC_ALIGN_BYTES=16)
ament_target_dependencies(rgbd_node rclcpp sensor_msgs geometry_msgs nav_msgs std_msgs cv_bridge message_filters std_srvs tf2_ros)
target_include_directories(rgbd_node PRIVATE "${ORB_SLAM3_ROOT}" "${ORB_SLAM3_ROOT}/include" "${ORB_SLAM3_ROOT}/include/CameraModels" "${ORB_SLAM3_ROOT}/Thirdparty/Sophus" ${OpenCV_INCLUDE_DIRS} ${Pangolin_INCLUDE_DIRS})
target_link_libraries(rgbd_node "${ORB_SLAM3_ROOT}/lib/libORB_SLAM3.so" Eigen3::Eigen ${OpenCV_LIBS} ${Pangolin_LIBRARIES})
install(TARGETS rgbd_node DESTINATION lib/${PROJECT_NAME})
install(PROGRAMS scripts/dense_node.py scripts/reload_map.py scripts/dense_mapping.py DESTINATION lib/${PROJECT_NAME})
install(FILES scripts/ros_cloud.py DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY rviz DESTINATION share/${PROJECT_NAME})
install(DIRECTORY launch DESTINATION share/${PROJECT_NAME})
ament_package()
